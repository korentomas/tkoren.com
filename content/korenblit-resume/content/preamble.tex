% ─────────────────────────────────────────────────────────

%  Minimal academic style: black only, left-margin section
%  labels, serif body. Compile with: pdflatex korenblit-resume.tex
% ─────────────────────────────────────────────────────────
\documentclass[11pt,letterpaper]{article}

% ─── Geometry: wide left margin holds the section labels ──
\usepackage[top=0.7in, bottom=0.45in, left=1.9in, right=0.65in]{geometry}
\linespread{0.98}
\newlength{\labelcol}\setlength{\labelcol}{1.0in}    % label box width (fits "Publications")
\newlength{\labelgap}\setlength{\labelgap}{0.25in}   % gap label -> text
\newlength{\leftpull}\setlength{\leftpull}{1.25in}   % labelcol + labelgap

% ─── Fonts ───────────────────────────────────────────────
\usepackage[T1]{fontenc}
\usepackage[utf8]{inputenc}
\usepackage{lmodern}           % Latin Modern serif (classic academic look)
\usepackage{microtype}

% ─── Links (no color) ────────────────────────────────────
\usepackage[hidelinks]{hyperref}

% ─── Layout ──────────────────────────────────────────────
\pagestyle{empty}
\setlength{\parindent}{0pt}
\setlength{\parskip}{0pt}

% ─── Section: label sits in the left margin (llap), text
%     flows in the main column. Page-break safe. ───────────
\renewcommand{\section}[1]{%
  \par\addvspace{5pt}%
  \noindent\llap{\smash{\parbox[t]{\labelcol}{\raggedright\scshape #1}}\hspace{\labelgap}}%
  \nopagebreak\ignorespaces}

% ─── Lists: plain dash bullets, tight ────────────────────
\usepackage{enumitem}
\setlist[itemize]{nosep, leftmargin=11pt, label={\textendash}, itemsep=1pt, topsep=1pt}

% ─── Helpers ─────────────────────────────────────────────
\newcommand{\daterange}[1]{{\small #1}}          % dates, plain
\newcommand{\sep}{\,\textbar\,}                   % inline separator: thin bar
\newcommand{\mono}[1]{#1}                          % no mono font; identity

% ─── Full-width header block (name + rule) pulled to the
%     far-left label column. ────────────────────────────────
\newcommand{\headerblock}[2]{%
  \noindent\hspace*{-\leftpull}%
  \begin{minipage}{\dimexpr\textwidth+\leftpull\relax}%
    {\fontsize{15}{18}\selectfont\textbf{#1}}\\[2pt]%
    {\small #2}\par\vspace{3pt}%
    \hrule height 0.4pt%
  \end{minipage}\par\vspace{2pt}}
\newcommand{\headeronly}[1]{%
  \noindent\hspace*{-\leftpull}%
  \begin{minipage}{\dimexpr\textwidth+\leftpull\relax}%
    {\fontsize{14}{17}\selectfont\textbf{#1}}\par\vspace{7pt}%
    \hrule height 0.4pt%
  \end{minipage}\par\vspace{2pt}}
